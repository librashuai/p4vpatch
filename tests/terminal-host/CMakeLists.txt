cmake_minimum_required(VERSION 3.21)
project(p4v_terminal_host_test LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This test requires Windows x64 and MSVC")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core Widgets WebChannel WebEngineWidgets)
add_executable(mock-host mock_host.cpp ../../src/terminal.cpp ../../src/preferences.cpp ../../src/qt_focus_hook.cpp ../../src/terminal_resources.qrc)
target_compile_definitions(mock-host PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(mock-host PRIVATE Qt6::Core Qt6::Widgets Qt6::WebChannel Qt6::WebEngineWidgets)
