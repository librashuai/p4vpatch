cmake_minimum_required(VERSION 3.21)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" P4VPATCH_VERSION LIMIT_COUNT 1)
if(NOT P4VPATCH_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)$")
    message(FATAL_ERROR "VERSION must contain MAJOR.MINOR.PATCH")
endif()
project(p4vpatch VERSION ${P4VPATCH_VERSION} LANGUAGES CXX RC)

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT MSVC)
    message(FATAL_ERROR "P4V patch and GUI launcher require MSVC x64 on Windows")
endif()
foreach(component PROJECT_VERSION_MAJOR PROJECT_VERSION_MINOR PROJECT_VERSION_PATCH)
    if(${component} GREATER 65535)
        message(FATAL_ERROR "Version components must fit Windows version resources (0..65535)")
    endif()
endforeach()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(/utf-8)
set(P4VPATCH_REPOSITORY "librashuai/p4vpatch" CACHE STRING "Public GitHub repository used for updates")
if(NOT P4VPATCH_REPOSITORY MATCHES "^[A-Za-z0-9_.-]+/[A-Za-z0-9_.-]+$")
    message(FATAL_ERROR "P4VPATCH_REPOSITORY must be owner/repository")
endif()
configure_file(src/version.h.in generated/version.h @ONLY)

# Compile with the public MSVC SDK; deploy beside P4V and use its Qt 6.8.6 runtime.
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core Widgets Network)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
add_library(launcher-update STATIC src/launcher_update.cpp)
target_include_directories(launcher-update PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(launcher-update PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(launcher-update PUBLIC Qt6::Core version)

add_library(launcher-theme STATIC src/launcher_theme.cpp)
target_include_directories(launcher-theme PUBLIC src)
target_compile_definitions(launcher-theme PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(launcher-theme PUBLIC Qt6::Widgets PRIVATE dwmapi)

add_executable(p4vpatch-launcher WIN32 src/launcher.cpp src/launcher_process.cpp src/launcher.rc src/version.rc src/launcher_resources.qrc)
target_link_libraries(p4vpatch-launcher PRIVATE launcher-update launcher-theme Qt6::Widgets Qt6::Network user32)

option(BUILD_PATCH "Build the injected Qt module" ON)
if(BUILD_PATCH)
    find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS WebChannel WebEngineWidgets)
    add_library(p4vpatch MODULE src/patch.cpp src/background_login.cpp src/preferences.cpp src/terminal.cpp src/qt_focus_hook.cpp src/terminal_resources.qrc src/version.rc)
    target_include_directories(p4vpatch PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_compile_definitions(p4vpatch PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(p4vpatch PRIVATE Qt6::Core Qt6::Widgets Qt6::WebChannel Qt6::WebEngineWidgets)
    set_target_properties(p4vpatch PROPERTIES PREFIX "" OUTPUT_NAME "p4vpatch")
    option(BUILD_LOGIN_TESTS "Build offline background login tests" OFF)
    if(BUILD_LOGIN_TESTS)
        enable_testing()
        add_subdirectory(tests/login)
    endif()
endif()
option(BUILD_LAUNCHER_TESTS "Build offline update tests" OFF)
if(BUILD_LAUNCHER_TESTS)
    enable_testing()
    add_subdirectory(tests/launcher)
endif()
